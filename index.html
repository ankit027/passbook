How do I link a Google Sheet? (one-time, 2 minutes)

1. Create a new Google Sheet. Open Extensions → Apps Script.

2. Delete any starter code and paste this in:

const SHEET_NAME = "Transactions"; function doGet() { return ContentService .createTextOutput("Passbook Google Sheet API is working") .setMimeType(ContentService.MimeType.TEXT); } function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { throw new Error("No data received"); } const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Record ID","Date","Type","Category","Payment Method","Amount", "Vehicle","Odometer (Km)","Difference (Km)","Planned Odometer", "Actual Odometer","Maintenance Difference","Source","Note","Timestamp" ]); } const records = Array.isArray(data) ? data : (data.entries || [data]); records.forEach(function(r) { const recordId = r.id || ("REC-" + Date.now() + "-" + Math.random().toString(36).substring(2,8)); const lastRow = sheet.getLastRow(); let existingRow = -1; if (lastRow > 1) { const ids = sheet.getRange(2,1,lastRow-1,1).getValues().flat(); existingRow = ids.indexOf(recordId) + 2; } const row = [ recordId, r.date || "", r.type || "Expense", r.category || (r.type === "Income" ? "Income" : ""), r.paymentMethod || "", Number(r.amount || 0), r.vehicle || "", r.odometer || "", r.difference || "", r.plannedOdo || "", r.actualOdo || "", (r.actualOdo !== undefined && r.plannedOdo !== undefined && r.actualOdo !== "" && r.plannedOdo !== "") ? Number(r.actualOdo) - Number(r.plannedOdo) : "", r.source || "", r.note || "", new Date() ]; if (existingRow > 1) { sheet.getRange(existingRow,1,1,row.length).setValues([row]); } else { sheet.appendRow(row); } }); sheet.getRange("F:F").setNumberFormat('₹#,##0.00'); return ContentService .createTextOutput(JSON.stringify({ success: true, message: records.length + " record(s) synced" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: String(err) })) .setMimeType(ContentService.MimeType.JSON); } }

3. Click Deploy → New deployment → Web app. Set "Execute as" to Me and "Who has access" to Anyone. Click Deploy and allow the permissions it asks for.

4. Copy the Web app URL it gives you and paste it into the field above, then click Save link.

Every entry you add can now be sent straight to the "Transactions" tab in that sheet. Because Google Apps Script doesn't send back a readable response to a browser page like this one, the button won't show a confirmation — check the sheet directly to see the rows land.
